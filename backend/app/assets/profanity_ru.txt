# Словарь маскирования нецензурной лексики в стенограмме (services/profanity.py). Правится без изменения кода и ASR.
# Слово сначала нормализуется: нижний регистр, «ё» → «е», латинские буквы-двойники → кириллица (a→а, e→е, o→о, p→р, c→с, x→х, y→у, k→к, m→м, t→т, h→н, b→в),
# три и более одинаковых буквы подряд → одна («бляяя» → «бля»).
#
# prefix NAME = регулярное выражение — набор приставок, подставляется в правила как {NAME}
# match = регулярное выражение — нормализованное слово целиком или по началу (якоря ^ и $ ставятся явно); совпадение — слово скрывается
# except = слова через пробел — никогда не скрываются (обычные слова с похожими сочетаниями букв)

prefix P = (?:а|без|бес|вз|вс|вы|до|за|из|ис|на|над|не|недо|ни|об|обо|от|ото|пере|по|под|пре|при|про|раз|рас|со|у|о|с|в|въ|съ|объ|отъ|подъ|разъ|изъ|долбо|полу){0,2}
prefix E = (?:вы|до|за|на|недо|об|обо|объ|от|отъ|пере|по|под|подъ|при|про|раз|разъ|съ|въ|изъ|у|уе|ни|долбо|пере|недо|рас|вз|вз|из){0,2}

# хуй и производные
match = ^{P}ху[йеяюи]
# пизда и производные (в т.ч. «пезд-»)
match = п[иея]зд
# ебать и производные (без однобуквенных приставок «в», «с», «о»: «вебинар», «себе», «оебь…»)
match = ^{E}еб(?:а|у|и|е|л|н|ы|ь|ш|к|о|ищ|ырь|уч|$)
match = ^еб$
# блядь и производные («бляха» — исключение)
match = ^(?:вы|на|за|по|раз|рас|под)?бля(?:$|д|т|ск|ш|х[ау]?$)
# мудак и производные («мудрый» не задевается)
match = ^муд(?:ак|ач|ил|оз|охат|ень|ня)
# манда — только формы самого слова («команда», «мандарин», «мандат» не задеваются)
match = ^манд(?:а|ы|е|у|ой|ою|ец|юк|ятин)$
match = залуп
match = г[ао]ндон
match = ^пид(?:ор|ар|р)|педераст|педерас
match = ^педик(?:а|у|ом|и|ов|ами)?$
match = ^с?сук(?:а|и|е|у|ой|ою|ам|ами|ах|ин|ина|ины|ину|ин[оа]?й)$
match = ^суч(?:ар|онок|онк|ка$|ку$|кой$|ий$|ья$|ье$|ьи$)
match = шлюх
match = дроч
match = ^(?:на|по|о|за)?хер(?:$|н|ов|ач|е[тн]|ь|ом$|а$|у$)

except = бляха бляхи бляху бляхой себе себя небе вебе веба вебинар вебинара вебинаре сук сучья сучок сучки сучку сучьев тебе
except = страхуй застрахуй застрахуйте перестрахуй команда мандат мандарин хулиган
